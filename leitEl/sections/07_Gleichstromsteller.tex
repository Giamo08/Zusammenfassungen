\section{Gleichstromsteller (Chopper)}

Ein Gleichstromsteller wandelt eine konstante Gleichspannung in eine \textbf{veränderliche Gleichspannung}.
Die Stellgrösse ist der \textbf{Tastgrad} $d$.

\subsection{Grundprinzip}

% --- bestehende Grafik: Grundschaltung Chopper ---
\begin{center}
    \includegraphics[width=0.85\linewidth]{images/Gleichstromsteller.png}
\end{center}
Mittelwert:
\[
\cbl{u_d = d \, U_{DC}}
\qquad
\cbl{d = \frac{t_{\mathrm{on}}}{T}}
\]

\subsection{Einquadrantensteller (1Q)}

% --- bestehende Grafik: 1Q ---
\begin{center}

    \includegraphics[width=0.85\linewidth]{images/Gleichstromsteller1Q.png}

\end{center}
Eigenschaften:
\begin{itemize}
    \item $u_d \ge 0$, $i_d \ge 0$
    \item Motorbetrieb vorwärts
\end{itemize}

\subsection{Zweiquadrantensteller (2Q)}

% --- bestehende Grafik: 2Q ---
\begin{center}
    \includegraphics[width=0.85\linewidth]{images/Gleichstromsteller2Q.png}
\end{center}
Eigenschaften:
\begin{itemize}
    \item $u_d \ge 0$, $i_d$ vorzeichenwechselnd
    \item Rekuperationsbetrieb möglich
\end{itemize}

\subsection{Vierquadrantensteller (4Q)}

% --- bestehende Grafik: 4Q ---
\begin{center}
    \includegraphics[width=0.85\linewidth]{images/Gleichstromsteller4Q.png}
\end{center}
Eigenschaften:
\begin{itemize}
    \item Alle Kombinationen von $u_d$ und $i_d$
    \item Vollständiger Motorbetrieb in allen Quadranten
\end{itemize}

\subsection{Merksätze}

\begin{itemize}
    \item Stellgrösse beim Gleichstromsteller ist der \textbf{Tastgrad $d$}.
    \item Nicht mit Zündwinkel $\alpha$ verwechseln.
\end{itemize}
